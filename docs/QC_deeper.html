process_motif_data <- function(file_list) {
  # Process files and calculate totals
  combined_data <- map_df(file_list, ~{
    message("\nProcessing: ", normalizePath(.x))
    
    # Extract sample name
    sample_name <- basename(.x) %>%
      str_remove(".genome.tsv_motif_summary_read\\d+_mepct\\d+\\.tsv$") %>%
      str_remove("241217AVITI-He-lab-24JX-") %>%
      str_remove("^_+") %>%
      str_remove("_+$")
    
    # Read and process data
    sample_data <- read_table(.x, 
                             col_names = c("motif", "count"), 
                             col_types = "cn",
                             skip_empty_rows = TRUE)
    
    total_reads <- sum(sample_data$count)
    
    processed <- sample_data %>%
      mutate(
        sample = sample_name,
        total_count = total_reads,
        percentage = (count / total_reads) * 100
      ) %>%
      select(sample, motif, percentage, total_count)
    
    return(processed)
  })
  
  # Create formatted labels and order data
  combined_data %>%
    mutate(
      sample_label = paste0(sample, " (n=", 
                           format(total_count, big.mark = ",", scientific = FALSE), ")"),
      sample_label = fct_reorder(sample_label, total_count)
    ) %>%
    arrange(desc(total_count)) %>%
    mutate(sample_label = factor(sample_label, levels = unique(sample_label)))
}

create_motif_plot <- function(processed_motif_data, 
                             title = "Motif Distribution with Total Read read_depth",
                             x_lab = "Percentage (%)",
                             y_lab = "Sample (Total site read_depth passing filter)",
                             point_size = 3,
                             text_size = 3) {
  ggplot(processed_motif_data, aes(x = percentage, y = sample_label)) +
    geom_point(
      aes(color = motif),
      size = point_size,
      position = position_jitterdodge(
        jitter.width = 0.2,
        jitter.height = 0.1,
        dodge.width = 0.7
      ),
      show.legend = FALSE
    ) +
    geom_text(
      aes(label = motif, color = motif),
      position = position_jitterdodge(
        jitter.width = 0.2,
        jitter.height = 0.1,
        dodge.width = 0.7
      ),
      size = text_size,
      vjust = -1.5,
      show.legend = FALSE
    ) +
   # scale_color_igv(guide = "none") +
    labs(
      title = title,
      x = x_lab,
      y = y_lab
    ) +
    theme_minimal() +
    theme(
      panel.grid.major.y = element_line(color = "grey90"),
      axis.text.y = element_text(size = 9, face = "bold"),
      axis.text.x = element_text(size = 10),
      plot.title = element_text(hjust = 0.5, size = 14)
    )
}



count_overlap <- function(file_list) {
  # Initialize an empty list to store the 'name' columns
  name_list <- list()
  
  # Read each file and store the 'name' column
  for(file in file_list){
    df_site <- read.table(file, header = T)
    df_site$name <- paste0(df_site$ref, "_", df_site$pos)
    name_list <- append(name_list, list(df_site$name))
  }
  
  # Find the overlap among all 'name' columns
  overlap <- Reduce(intersect, name_list)
  
  # Return the overlap count
  return(length(overlap))
}
