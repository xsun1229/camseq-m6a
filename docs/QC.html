avg_length_all <- c()
for (file in json_files) {
  # Read the JSON file
  
  json_data <- fromJSON(file)
  
  total_bases <- json_data$basepair_counts$output
  output_reads <- json_data$read_counts$output
  
  avg_length <- round(total_bases / (output_reads * 2),digits = 4)  # Paired-end
  sample <- extracted_pattern <- sub(".*24JX-([0-9-]+)_run1.*", "\\1", file)
  
  tmp <- c(sample, avg_length)
  avg_length_all <- rbind(avg_length_all,tmp)
  
}

folder_countsites <- "/project/xinhe/xsun/camseq/data/workspace/count_sites/pileup/"
files_countsites <- list.files(path = folder_countsites,pattern = "genome.tsv",full.names = T)

p_me_dist <- list()
p_cov_dist <- list()
strand_bias_all <- c()
for (file in files_countsites) {
  
  sample <- regmatches(basename(file),
            regexpr("\\d+-\\d+-\\d+(?=\\.)", basename(file), perl = TRUE))
  
  df <- data.table::fread(file)
  
  # Calculate methylation and filter
  df <- df %>%
    mutate(
      total = convertedBaseCount + unconvertedBaseCount,
      methylation_pct = (unconvertedBaseCount / total) * 100
    ) %>%
    filter(total >= 10)   # Remove low-coverage sites
  
  if (nrow(df) > 0) {
    strand_bias <- df %>%
    group_by(strand) %>%
    summarise(
      mean_methylation = round(mean(methylation_pct, na.rm = TRUE),digits = 4),
      n_sites_pass_filtering = n()
    ) %>%
    mutate(sample = sample) %>%          # Add sample column
    select(sample, everything())         # Reorder columns (sample first)
  }else{
     strand_bias <- data.frame(sample = c(sample,sample),
                               strand = c("+","-"),
                               mean_methylation = c("no site passing filtering","no site passing filtering"),
                               n_sites_pass_filtering = c("no site passing filtering","no site passing filtering"))
    
  }
  
  strand_bias_all <- rbind(strand_bias_all,strand_bias)
  
  p_me_dist[[sample]] <- ggplot(df, aes(x=methylation_pct)) +
    geom_histogram(bins=30, fill="blue", alpha = 0.6) +
    labs(x="Methylation (%)", y="Number of sites") +
    ggtitle(sample) +
    theme_minimal()
  
  p_cov_dist[[sample]] <- ggplot(df, aes(x=total)) +
    geom_histogram(bins=50, fill="blue", alpha = 0.6) +
    #xlim(-1, 100) +
    labs(x="Total reads per site", y="Count") +
    ggtitle(sample) +
    theme_minimal()


  
  
  
}


combined <- arrangeGrob(
  grobs = p_me_dist,
  nrow = 2,
  top = textGrob("Methylation Distribution", gp = gpar(fontsize = 14))
)
grid.draw(combined)

combined <- arrangeGrob(
  grobs = p_cov_dist,
  nrow = 2,
  top = textGrob("Coverage Distribution", gp = gpar(fontsize = 14))
)
grid.draw(combined)
