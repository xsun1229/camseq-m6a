json_folder <- "/project/xinhe/xsun/camseq/data/workspace/report_reads/trimming/"

# List all JSON files in the folder
json_files <- list.files(json_folder, pattern = "*.json", full.names = TRUE)

avg_length_all <- c()
for (file in json_files) {
  # Read the JSON file
  
  json_data <- fromJSON(file)
  
  total_bases <- json_data$basepair_counts$output
  output_reads <- json_data$read_counts$output
  
  avg_length <- round(total_bases / (output_reads * 2),digits = 4)  # Paired-end
  sample <- extracted_pattern <- sub(".*24JX-([0-9-]+)_run1.*", "\\1", file)
  
  tmp <- c(sample, avg_length)
  avg_length_all <- rbind(avg_length_all,tmp)
  
}

rownames(avg_length_all) <- NULL
colnames(avg_length_all) <- c("sample","avg_length/read")
avg_length_all <- as.data.frame(avg_length_all )

DT::datatable(avg_length_all,caption = htmltools::tags$caption(style = 'caption-side: left; text-align: left; color:black;  font-size:150% ;','Average length after trimming'),options = list(pageLength = 10) )

counts <- 10
pct <- 10

folder_countsites <- "/project/xinhe/xsun/camseq/data/workspace/count_sites/pileup/"

file_list <- list.files(path = folder_countsites,pattern = paste0("counts_cutoff",counts,"_pct",pct),full.names = T)
file_list <- file_list[c(1,2,7,8)]

# Process files and calculate totals
combined_data <- map_df(file_list, ~{
  # Extract sample name
  sample_name <- str_remove(.x, "_motif_counts_cutoff\\d+_pct\\d+.tsv$") %>%
    str_remove(".*241217AVITI-He-lab-24JX-") %>%
    str_remove("^/+") %>%
    str_remove("/+$")
  
  # Read and process data
  sample_data <- read_tsv(.x) %>% 
    filter(motif != "motif") %>%
    mutate(
      motif = str_replace(motif, "(.)([+-])(.)", "\\2\\1A\\3")
    )
  
  total_reads <- sum(sample_data$count)
  
  sample_data %>%
    mutate(
      sample = sample_name,
      total_count = total_reads,
      percentage = (count / total_reads) * 100
    ) %>%
    select(sample, motif, percentage, total_count)
})

# Create formatted labels ordered by total count
combined_data <- combined_data %>%
  mutate(
    sample_label = paste0(sample, " (n=", format(total_count, big.mark = ",", scientific = FALSE), ")"),
    sample_label = fct_reorder(sample_label, total_count)
  )

# Create visualization
ggplot(combined_data, aes(x = percentage, y = sample_label)) +
  geom_point(
    aes(color = motif),
    size = 3,
    position = position_jitterdodge(
      jitter.width = 0.2,
      jitter.height = 0.1,
      dodge.width = 0.7
    ),
    show.legend = FALSE
  ) +
  geom_text(
    aes(label = motif, color = motif),
    position = position_jitterdodge(
      jitter.width = 0.2,
      jitter.height = 0.1,
      dodge.width = 0.7
    ),
    size = 3,
    vjust = -1.5,
    show.legend = FALSE
  ) +
  scale_color_igv(guide = "none") +
  labs(
    title = "Motif Distribution with Total Read Counts",
    x = "Percentage (%)",
    y = "Sample (Total site counts passing filter)",
    caption = "-/+ represent strands"
  ) +
  theme_minimal() +
  theme(
    panel.grid.major.y = element_line(color = "grey90"),
    axis.text.y = element_text(size = 9, face = "bold"),
    axis.text.x = element_text(size = 10),
    plot.title = element_text(hjust = 0.5, size = 14)
  )
