Comparison

We vary the read depth (10, 15, 20)/ methylation level (0, 0.05,0.1) / p-value (0.01, 0.001, 0.0001) for our data, then compare the m6a sites with the paper.

READDEPTH_vec <- c(10, 15, 20)
p_cutoff_str_vec <- c("0.01", "0.001", "0.0001")
methylation_vec <- c(0, 0.05, 0.1)

df_summary <- c()
for (rd in READDEPTH_vec) {
  
  for (p_cutoff_str in p_cutoff_str_vec){
    
    for (methylation_level in methylation_vec){
      
      file_ours <- paste0("/project/xinhe/xsun/camseq/data/workspace_round3/count_sites/pileup/processed_binomp_", p_cutoff_str,"_1n/sample05_with_id_read", rd,"_binomp", p_cutoff_str,".tsv")
      df_ours_raw <- fread(file_ours)
      df_ours_raw$methylation_mepct <- df_ours_raw$methylation_mepct/100
      df_ours_mefiltered <- df_ours_raw[df_ours_raw$methylation_mepct >= methylation_level,]
      
      df_ours_mefiltered$site <- paste0(df_ours_mefiltered$ref,"-",df_ours_mefiltered$pos)
      df_paper_raw$site <- paste0(df_paper_raw$Chrom,"-",df_paper_raw$Pos)
      
      num_overlap <- sum(df_ours_mefiltered$site %in% df_paper_raw$site)
      
      tmp <- c(rd, p_cutoff_str, methylation_level, nrow(df_ours_mefiltered), num_overlap, num_overlap/nrow(df_ours_mefiltered)*100)
      df_summary <- rbind(df_summary, tmp)
      
    }
    
    
    
  }
  
  
}

df_summary <- as.data.frame(df_summary)
colnames(df_summary) <- c("Read_depth", "binomial_p","methylation_level_fileter","#sites_filtered","#sites_overlap_with_paper","%sites_overlap_with_paper")
